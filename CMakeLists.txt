cmake_minimum_required(VERSION 3.20)
project(dsp3D VERSION 2.0.0 LANGUAGES C)
option(DSP3D_BUILD_TESTS "Build native correctness tests and reference renderer" ON)
add_library(dsp3d STATIC src/dsp3D.c src/dsp3D_model.c)
target_include_directories(dsp3d PUBLIC inc)
target_compile_features(dsp3d PUBLIC c_std_11)
if(CMAKE_C_COMPILER_ID MATCHES "GNU|Clang")
    target_compile_options(dsp3d PRIVATE -Wall -Wextra -Wpedantic -ffp-contract=fast -fno-strict-aliasing)
    target_link_libraries(dsp3d PUBLIC m)
endif()
if(DSP3D_BUILD_TESTS)
    enable_testing()
    add_executable(dsp3d_tests tests/test_renderer.c assets/generated/metamask.c assets/generated/suzanne.c)
    target_include_directories(dsp3d_tests PRIVATE assets/generated)
    if(MSVC)
        target_compile_options(dsp3d_tests PRIVATE /UNDEBUG)
    else()
        target_compile_options(dsp3d_tests PRIVATE -UNDEBUG -Wall -Wextra -Wpedantic)
    endif()
    target_link_libraries(dsp3d_tests PRIVATE dsp3d)
    add_test(NAME renderer COMMAND dsp3d_tests)
    set_tests_properties(renderer PROPERTIES WORKING_DIRECTORY "${CMAKE_SOURCE_DIR}")
    add_executable(dsp3d_preview examples/desktop/preview.c assets/generated/metamask.c assets/generated/suzanne.c)
    target_include_directories(dsp3d_preview PRIVATE assets/generated)
    target_link_libraries(dsp3d_preview PRIVATE dsp3d)
endif()
